<script>
/* ============================================================
   HTML-локер + экран оплаты
   ПАРОЛЬ: 1937
   ============================================================ */

const PASSWORD       = '1937';   // <-- ПАРОЛЬ РАЗБЛОКИРОВКИ
const PAY_AMOUNT     = '1937';   // ожидаемая сумма на экране оплаты
const COUNTDOWN_FROM = 10;       // отсчёт до активации
const VIBRATE_MS     = 400;

const el = {
  lock:      document.getElementById('lock'),
  pay:       document.getElementById('pay'),
  countdown: document.getElementById('countdown'),
  barfill:   document.getElementById('barfill'),
  code:      document.getElementById('code'),
  err:       document.getElementById('err'),
  unlockBtn: document.getElementById('unlockBtn'),
  toPayBtn:  document.getElementById('toPayBtn'),
  payAmount: document.getElementById('payAmount'),
  payBtn:    document.getElementById('payBtn'),
  payStatus: document.getElementById('payStatus'),
  backBtn:   document.getElementById('backBtn'),
};

let payConfirmed = false;

/* ---------- вспомогательные ---------- */

function vibrate(pattern) {
  if (navigator.vibrate) navigator.vibrate(pattern);
}

function beep(freq, dur) {
  try {
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'square';
    osc.frequency.value = freq;
    gain.gain.value = 0.07;
    osc.connect(gain); gain.connect(ctx.destination);
    osc.start();
    setTimeout(() => { osc.stop(); ctx.close(); }, dur);
  } catch (e) {}
}

function fullscreen() {
  const d = document.documentElement;
  if (d.requestFullscreen) d.requestFullscreen().catch(() => {});
  else if (d.webkitRequestFullscreen) d.webkitRequestFullscreen();
}

/* ---------- обратный отсчёт ---------- */

let n = COUNTDOWN_FROM;
el.countdown.textContent = n;

const timer = setInterval(() => {
  n--;
  el.countdown.textContent = Math.max(n, 0);
  el.barfill.style.width = ((COUNTDOWN_FROM - n) / COUNTDOWN_FROM * 100) + '%';
  vibrate(40);
  beep(500 + n * 40, 70);

  if (n <= 0) {
    clearInterval(timer);
    el.countdown.textContent = '0';
    el.lock.classList.add('glitch');
    el.toPayBtn.style.display = 'block';   // показать кнопку оплаты
    vibrate([300, 80, 300, 80, 600]);
    beep(180, 400);

    setInterval(() => {
      vibrate(VIBRATE_MS);
      beep(160, 120);
    }, 1100);
  }
}, 1000);

/* ---------- переход к оплате ---------- */

el.toPayBtn.addEventListener('click', () => {
  el.lock.style.display = 'none';
  el.pay.style.display = 'flex';
  el.payAmount.focus();
  vibrate(60);
});

/* ---------- подтверждение оплаты ---------- */

el.payBtn.addEventListener('click', () => {
  const val = (el.payAmount.value || '').replace(/\s/g, '');
  if (val === PAY_AMOUNT) {
    payConfirmed = true;
    el.payStatus.style.color = '#0f0';
    el.payStatus.textContent = 'ПЛАТЁЖ ПОДТВЕРЖДЁН. ВВЕДИТЕ КОД.';
    beep(900, 150);
    vibrate(80);
    setTimeout(() => {
      el.pay.style.display = 'none';
      el.lock.style.display = 'flex';
      el.code.focus();
    }, 900);
  } else {
    el.payStatus.style.color = '#f00';
    el.payStatus.textContent = 'НЕВЕРНАЯ СУММА';
    vibrate([120, 80, 120]);
    beep(120, 180);
    el.payAmount.value = '';
    el.payAmount.focus();
  }
});

/* ---------- возврат на локер ---------- */

el.backBtn.addEventListener('click', () => {
  el.pay.style.display = 'none';
  el.lock.style.display = 'flex';
  el.code.focus();
});

/* ---------- проверка пароля ---------- */

function checkCode() {
  const val = (el.code.value || '').trim();
  if (val === PASSWORD) {
    el.err.style.color = '#0f0';
    el.err.textContent = 'ДОСТУП РАЗРЕШЁН';
    beep(900, 150);
    vibrate(80);
    setTimeout(() => {
      el.lock.style.display = 'none';
      el.pay.style.display = 'none';
      document.body.style.background = '#000';
      history.pushState(null, '', location.href);
    }, 700);
  } else {
    el.err.style.color = '#f00';
    el.err.textContent = 'НЕВЕРНЫЙ КОД';
    vibrate([120, 80, 120]);
    beep(120, 180);
    el.code.value = '';
    el.code.focus();
  }
}

el.unlockBtn.addEventListener('click', checkCode);
el.code.addEventListener('keydown', (e) => {
  if (e.key === 'Enter') checkCode();
});

/* ---------- блокировка выхода ---------- */

history.pushState(null, '', location.href);
window.addEventListener('popstate', () => {
  history.pushState(null, '', location.href);
});

window.addEventListener('beforeunload', (e) => {
  e.preventDefault();
  e.returnValue = '';
});

window.addEventListener('contextmenu', (e) => e.preventDefault());
window.addEventListener('selectstart', (e) => e.preventDefault());

window.addEventListener('keydown', (e) => {
  if ((e.ctrlKey || e.metaKey) && (e.key === 'w' || e.key === 't' || e.key === 'n')) {
    e.preventDefault();
  }
  if (e.key === 'F5') e.preventDefault();
});

/* ---------- запуск ---------- */

function start() {
  fullscreen();
  el.code.focus();
}
document.body.addEventListener('touchstart', function once() {
  document.body.removeEventListener('touchstart', once);
  start();
}, { once: true });
if (!('ontouchstart' in window)) start();
